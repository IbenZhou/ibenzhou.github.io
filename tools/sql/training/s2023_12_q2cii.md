prompt: |
  Return the **names** of every horse whose level is **intermediate** or **beginner**.
seed: |
  CREATE TABLE Horse (
    HorseID INTEGER PRIMARY KEY,
    Name VARCHAR(30),
    Height REAL,
    Age INTEGER,
    HorseLevel VARCHAR(15)
  );
  INSERT INTO Horse VALUES
    (1, 'Bramble', 1.52, 7, 'intermediate'),
    (2, 'Clover', 1.44, 11, 'beginner'),
    (3, 'Dune', 1.61, 5, 'advanced'),
    (4, 'Ember', 1.49, 9, 'intermediate'),
    (5, 'Fern', 1.40, 14, 'beginner');
answer: |
  SELECT Name
  FROM Horse
  WHERE HorseLevel = 'intermediate' OR HorseLevel = 'beginner';
wrong: |
  SELECT Name
  FROM Horse
  WHERE HorseLevel = 'intermediate';

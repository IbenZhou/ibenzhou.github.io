prompt: |
  Return the customer's ID, their name and the **total cost** of the orders they have
  **not** paid for. Give the total a suitable name.
seed: |
  CREATE TABLE Customer (
    CustomerID INTEGER PRIMARY KEY,
    Name VARCHAR(40),
    Phone VARCHAR(20)
  );
  CREATE TABLE Orders (
    OrderID INTEGER PRIMARY KEY,
    CustomerID INTEGER REFERENCES Customer(CustomerID),
    TotalCost REAL,
    Paid BOOLEAN,
    OrderDate DATE,
    Complete BOOLEAN
  );
  INSERT INTO Customer VALUES
    (1, 'Aisha Bello', '555-0101'),
    (2, 'Marco Ruiz', '555-0102'),
    (3, 'Hana Ito', '555-0103');
  INSERT INTO Orders VALUES
    (10, 1, 40.00, FALSE, '2025-02-01', TRUE),
    (11, 1, 25.50, TRUE, '2025-02-03', TRUE),
    (12, 2, 18.75, FALSE, '2025-02-04', FALSE),
    (13, 2, 60.00, FALSE, '2025-02-09', TRUE),
    (14, 3, 12.00, TRUE, '2025-02-10', TRUE);
answer: |
  SELECT CUSTOMER.CustomerID, CUSTOMER.Name, SUM(Orders.TotalCost) AS UnpaidTotal
  FROM Customer
  INNER JOIN Orders ON Customer.CustomerID = Orders.CustomerID
  WHERE Orders.Paid = FALSE
  GROUP BY CUSTOMER.CustomerID, CUSTOMER.Name;

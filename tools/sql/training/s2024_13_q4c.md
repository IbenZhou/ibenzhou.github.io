prompt: |
  Return the **title** of each show and the **number of times** it is scheduled. Give
  the count a suitable name.
seed: |
  CREATE TABLE Show (
    ShowID VARCHAR(6) PRIMARY KEY,
    Title VARCHAR(50),
    Duration INTEGER
  );
  CREATE TABLE Performance (
    PerformanceID INTEGER PRIMARY KEY,
    ShowID VARCHAR(6) REFERENCES Show(ShowID),
    ShowDate DATE,
    StartTime TIME
  );
  INSERT INTO Show VALUES
    ('MK12', 'Midnight Kitchen', 95),
    ('TR03', 'The Rooftop', 80),
    ('GL07', 'Glasshouse', 110);
  INSERT INTO Performance VALUES
    (1, 'MK12', '2024-06-01', '19:30'),
    (2, 'MK12', '2024-06-08', '19:30'),
    (3, 'MK12', '2024-06-15', '14:00'),
    (4, 'TR03', '2024-06-02', '20:00'),
    (5, 'GL07', '2024-06-09', '18:00');
answer: |
  SELECT Show.Title, COUNT(Performance.PerformanceID) AS ScheduledTimes
  FROM Show
  INNER JOIN Performance ON Show.ShowID = Performance.ShowID
  GROUP BY Show.Title;

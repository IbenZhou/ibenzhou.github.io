  Add this car to the Car table: registration **123AA**, make **Tiger**, model

seed: |
  CREATE TABLE Shop (
    ShopID VARCHAR(10) PRIMARY KEY,
    ManagerID INTEGER,
    Address VARCHAR(50),
    Town VARCHAR(30),
    TelephoneNumber VARCHAR(20)
  );
  CREATE TABLE Car (
    RegistrationNumber VARCHAR(8) PRIMARY KEY,
    Make VARCHAR(20),
    Model VARCHAR(20),
    NumberOfMiles INTEGER,
    ShopID VARCHAR(10) REFERENCES Shop(ShopID)
  );
  INSERT INTO Shop VALUES
    ('12BSTREET', 1, '12 Bridge Street', 'Leeds', '0113 555 0101');
  INSERT INTO Car VALUES
    ('AA11BCD', 'Tiger', 'Lioness', 12000, '12BSTREET');

  INSERT INTO Car
  VALUES ('123AA', 'Tiger', 'Lioness', 10500, '12BSTREET');

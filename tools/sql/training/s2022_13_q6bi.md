title: How many devices one member of staff has

difficulty: advanced

coverage: SELECT, COUNT, INNER JOIN, WHERE
schema: |
  STAFF(StaffID, FirstName, LastName)
  DEVICE(DeviceID, Type, DatePurchased, StaffID)
prompt: |
  The school's IT register links every device to the member of staff who holds it.
  Return the **number of devices** held by the staff member whose first name is **Ali**
  and whose last name is **Khan**.
seed: |
  CREATE TABLE Staff (
    StaffID INTEGER PRIMARY KEY,
    FirstName VARCHAR(30),
    LastName VARCHAR(30)
  );
  CREATE TABLE Device (
    DeviceID INTEGER PRIMARY KEY,
    Type VARCHAR(20),
    DatePurchased DATE,
    StaffID INTEGER REFERENCES Staff(StaffID)
  );
  INSERT INTO Staff VALUES
    (1, 'Ali', 'Khan'),
    (2, 'Ali', 'Ng'),
    (3, 'Sara', 'Khan');
  INSERT INTO Device VALUES
    (100, 'laptop', '2024-02-01', 1),
    (101, 'tablet', '2024-03-11', 1),
    (102, 'laptop', '2023-09-20', 2),
    (103, 'projector', '2022-05-06', 3);
answer: |
  SELECT COUNT(Device.DeviceID)
  FROM Staff
  INNER JOIN Device ON Staff.StaffID = Device.StaffID
  WHERE Staff.FirstName = 'Ali' AND Staff.LastName = 'Khan';
wrong: |
  SELECT COUNT(Device.DeviceID)
  FROM Staff
  INNER JOIN Device ON Staff.StaffID = Device.StaffID
  WHERE Staff.FirstName = 'Ali' OR Staff.LastName = 'Khan';
check: rows

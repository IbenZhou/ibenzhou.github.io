  A complaints system records which member of staff handled each complaint. Create the

  and RemoteWorker (a yes / no value).

  CREATE TABLE Staff (
    StaffID INTEGER PRIMARY KEY,
    StaffFirstName VARCHAR(30),
    StaffLastName VARCHAR(30),
    Department VARCHAR(30),
    RemoteWorker BOOLEAN
  );

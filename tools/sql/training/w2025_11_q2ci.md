  PLACEMENT(PlacementID, StudentID, CompanyID, StartDate, EndDate, Complete)
prompt: |
  Work placements that have been completed are archived elsewhere. Write the statement
  that deletes **only** the completed placements from Placement.
seed: |
  CREATE TABLE Placement (
    PlacementID INTEGER PRIMARY KEY,
    StudentID VARCHAR(8),
    CompanyID VARCHAR(8),
    StartDate DATE,
    EndDate DATE,
    Complete BOOLEAN
  );
  INSERT INTO Placement VALUES
    (1, 'LDEA01', 'NEAM', '2024-06-01', '2024-06-28', TRUE),
    (2, 'LDEA01', 'NEAM', '2025-01-06', '2025-02-14', FALSE),
    (3, 'BRX777', 'HALO', '2024-09-02', '2024-09-27', TRUE);
answer: |
  DELETE FROM Placement
  WHERE Complete = TRUE;
wrong: |
  DELETE FROM Placement
  WHERE Complete = FALSE;

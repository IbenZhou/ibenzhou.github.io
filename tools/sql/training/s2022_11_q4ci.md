prompt: |
  Northbridge College records how each student performs in short tests. The Student and
  Test tables already exist. Create the table **Student_Test** to hold each result:
  StudentID (whole number), TestID (up to 4 characters) and Mark (a number that may have
  decimals). A student can sit the same test only once, so the **combination** of
  StudentID and TestID is the primary key. StudentID must reference Student(StudentID)
  and TestID must reference Test(TestID).
seed: |
  CREATE TABLE Student (
    StudentID INTEGER PRIMARY KEY,
    FirstName VARCHAR(30),
    LastName VARCHAR(30)
  );
  CREATE TABLE Test (
    TestID VARCHAR(4) PRIMARY KEY,
    Description VARCHAR(40),
    TotalMarks INTEGER
  );
answer: |
  CREATE TABLE Student_Test (
    StudentID INTEGER,
    TestID VARCHAR(4),
    Mark REAL,
    PRIMARY KEY (StudentID, TestID),
    FOREIGN KEY (StudentID) REFERENCES Student(StudentID),
    FOREIGN KEY (TestID) REFERENCES Test(TestID)
  );
wrong: |

  );

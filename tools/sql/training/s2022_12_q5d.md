prompt: |
  Return the **number of films** released during January 2022.
seed: |
  CREATE TABLE Film_Fact (
    FilmID INTEGER PRIMARY KEY,
    FilmTitle VARCHAR(60),
    ReleaseDate DATE,
    Category VARCHAR(20)
  );
  INSERT INTO Film_Fact VALUES
    (10, 'Cinderella', '2021-12-03', 'Fantasy'),
    (11, 'The Long Pier', '2022-01-21', 'Drama'),
    (12, 'Cinder Road', '2022-01-05', 'Drama'),
    (13, 'Salt and Iron', '2022-03-14', 'History'),
    (14, 'Night Bus', '2022-01-31', 'Thriller');
answer: |
  SELECT COUNT(FilmID)
  FROM Film_Fact
  WHERE ReleaseDate >= '2022-01-01' AND ReleaseDate <= '2022-01-31';

  FROM Film_Fact

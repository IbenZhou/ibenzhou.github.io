title: Paid invoices in 2023

difficulty: advanced
source: 同构于 2024 冬季 P12 Q6(b)
coverage: SELECT, SUM, WHERE
schema: |
  INVOICE(InvoiceID, SupplierID, Amount, Paid, DateSent)
prompt: |
  Return the **total amount** of every invoice sent during **2023** that has been paid.
seed: |
  CREATE TABLE Invoice (
    InvoiceID INTEGER PRIMARY KEY,
    SupplierID VARCHAR(8),
    Amount REAL,
    Paid BOOLEAN,
    DateSent DATE
  );
  INSERT INTO Invoice VALUES
    (1, 'SU01', 150.00, TRUE, '2023-01-15'),
    (2, 'SU01', 90.00, TRUE, '2023-11-30'),
    (3, 'SU02', 200.00, FALSE, '2023-06-10'),
    (4, 'SU02', 75.00, TRUE, '2024-01-05'),
    (5, 'SU03', 60.00, TRUE, '2022-12-31');
answer: |
  SELECT SUM(Amount)
  FROM Invoice
  WHERE Paid = TRUE AND DateSent >= '2023-01-01' AND DateSent <= '2023-12-31';
wrong: |
  SELECT SUM(Amount)
  FROM Invoice
  WHERE Paid = TRUE;
check: rows

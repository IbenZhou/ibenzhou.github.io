prompt: |
  For the character with the ID **0002**, change **Level** to **3** and **Money** to
  **10000.00**. Leave every other character unchanged.
seed: |
  CREATE TABLE GameCharacter (
    CharacterName VARCHAR(30),
    CharacterID VARCHAR(4) PRIMARY KEY,
    Username VARCHAR(20),
    Level INTEGER,
    Money REAL
  );
  INSERT INTO GameCharacter VALUES
    ('Kat', '0001', 'KAT123', 2, 1500),
    ('Kit', '0002', 'KAT123', 5, 300),
    ('Rex', '0003', 'ROB77', 7, 9000);
answer: |
  UPDATE GameCharacter
  SET Level = 3, Money = 10000.00
  WHERE CharacterID = '0002';
wrong: |
  UPDATE GameCharacter
  SET Level = 3, Money = 10000.00;

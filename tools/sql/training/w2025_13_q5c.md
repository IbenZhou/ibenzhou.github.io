prompt: |
  Return the **ProductID**, **ProductName** and **ComplaintDetails** for every product
  with a rating of **5 or less**, with the worst-rated products listed first.
seed: |
  CREATE TABLE Product (
    ProductID VARCHAR(8) PRIMARY KEY,
    ProductName VARCHAR(40),
    ProductDetail VARCHAR(60),
    Price REAL,
    Rating INTEGER
  );
  CREATE TABLE Complaint (
    ComplaintID INTEGER PRIMARY KEY,
    ProductID VARCHAR(8) REFERENCES Product(ProductID),
    CustomerID VARCHAR(8),
    ComplaintDetails VARCHAR(120),
    StaffID INTEGER
  );
  INSERT INTO Product VALUES
    ('P100', 'Kettle', '1.7 litre stainless', 29.99, 4),
    ('P101', 'Toaster', 'two slot', 24.50, 2),
    ('P102', 'Blender', '700W', 39.00, 8),
    ('P103', 'Iron', 'steam', 19.95, 5);
  INSERT INTO Complaint VALUES
    (1, 'P100', 'C1', 'Lid sticks', 7),
    (2, 'P101', 'C2', 'Burns one side', 7),
    (3, 'P102', 'C1', 'Too noisy', 8),
    (4, 'P103', 'C3', 'Water tank leaks', 9);
answer: |
  SELECT Product.ProductID, Product.ProductName, Complaint.ComplaintDetails
  FROM Product
  INNER JOIN Complaint ON Product.ProductID = Complaint.ProductID
  WHERE Product.Rating <= 5
  ORDER BY Product.Rating DESC;
wrong: |
  SELECT Product.ProductID, Product.ProductName, Complaint.ComplaintDetails
  FROM Product
  INNER JOIN Complaint ON Product.ProductID = Complaint.ProductID
  WHERE Product.Rating <= 5
  ORDER BY Product.Rating ASC;

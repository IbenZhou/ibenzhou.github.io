prompt: |
  Return each department's **name** and the **number of subject codes** it offers. Give
  the count a suitable name, and list the departments in ascending order of name.
seed: |
  CREATE TABLE Department (
    DepartmentID INTEGER PRIMARY KEY,
    DepartmentName VARCHAR(40),
    DepartmentLeader VARCHAR(40)
  );
  CREATE TABLE Subject (
    SubjectCode VARCHAR(8) PRIMARY KEY,
    SubjectName VARCHAR(40),
    DepartmentID INTEGER REFERENCES Department(DepartmentID)
  );
  INSERT INTO Department VALUES
    (1, 'Arts', 'Ms Ito'),
    (2, 'Sciences', 'Mr Okafor'),
    (3, 'Languages', 'Ms Duarte');
  INSERT INTO Subject VALUES
    ('ART01', 'Art', 1),
    ('MUS01', 'Music', 1),
    ('BIO01', 'Biology', 2),
    ('CHE01', 'Chemistry', 2),
    ('PHY01', 'Physics', 2);
answer: |
  SELECT Department.DepartmentName, COUNT(Subject.SubjectCode) AS NumberSubjectCodes
  FROM Department
  INNER JOIN Subject ON Department.DepartmentID = Subject.DepartmentID
  GROUP BY Department.DepartmentID, Department.DepartmentName
  ORDER BY Department.DepartmentName ASC;
wrong: |

  GROUP BY Department.DepartmentID, Department.DepartmentName
  ORDER BY Department.DepartmentName DESC;

title: Add an order date to purchases

schema: |
  PURCHASE(PurchaseID, ItemName, Quantity)
prompt: |
  Purchases now need to record when they were made. Add one field called **OrderDate**
  to Purchase; it stores a date.
seed: |
  CREATE TABLE Purchase (
    PurchaseID INTEGER PRIMARY KEY,
    ItemName VARCHAR(40),
    Quantity INTEGER
  );
  INSERT INTO Purchase VALUES (1, 'Exercise book', 30);
answer: |
  ALTER TABLE Purchase ADD COLUMN OrderDate DATE;
wrong: |
  ALTER TABLE Purchase ADD COLUMN OrderDate VARCHAR(20);
check: schema
checktable: Purchase

prompt: |
  A chain of car showrooms records which shop holds each car. Return each shop's ID and
  the **number of cars** it holds.
seed: |
  CREATE TABLE Shop (
    ShopID VARCHAR(10) PRIMARY KEY,
    ManagerID INTEGER,
    Address VARCHAR(50),
    Town VARCHAR(30),
    TelephoneNumber VARCHAR(20)
  );
  CREATE TABLE Car (
    RegistrationNumber VARCHAR(8) PRIMARY KEY,
    Make VARCHAR(20),
    Model VARCHAR(20),
    NumberOfMiles INTEGER,
    ShopID VARCHAR(10) REFERENCES Shop(ShopID)
  );
  INSERT INTO Shop VALUES
    ('12BSTREET', 1, '12 Bridge Street', 'Leeds', '0113 555 0101'),
    ('8MARKET', 2, '8 Market Row', 'York', '01904 555 0102');
  INSERT INTO Car VALUES
    ('AA11BCD', 'Tiger', 'Lioness', 12000, '12BSTREET'),
    ('CC22DEF', 'Tiger', 'Cub', 8000, '12BSTREET'),
    ('EE33FGH', 'Falcon', 'Swift', 21000, '8MARKET');
answer: |
  SELECT ShopID, COUNT(RegistrationNumber)
  FROM Car
  GROUP BY ShopID;
wrong: |
  SELECT ShopID, COUNT(RegistrationNumber)
  FROM Car;

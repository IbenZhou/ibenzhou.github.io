prompt: |
  Return each player's **name** and the **number of characters** they have created. Give
  the count a suitable name. List the players in **descending order of PlayerID**.
seed: |
  CREATE TABLE Player (
    PlayerID INTEGER PRIMARY KEY,
    PlayerName VARCHAR(30),
    DateJoined DATE
  );
  CREATE TABLE GameCharacter (
    CharacterID INTEGER PRIMARY KEY,
    CharacterName VARCHAR(30),
    DateCreated DATE,
    TheLevel INTEGER,
    NumberOfPoints INTEGER,
    PlayerID INTEGER REFERENCES Player(PlayerID)
  );
  INSERT INTO Player VALUES
    (1, 'Asha', '2025-05-01'),
    (2, 'Bilal', '2025-06-11'),
    (3, 'Cleo', '2025-07-20');
  INSERT INTO GameCharacter VALUES
    (101, 'Ash', '2025-05-02', 4, 120, 1),
    (102, 'Flint', '2025-05-09', 2, 40, 1),
    (103, 'Bolt', '2025-06-12', 6, 300, 2),
    (104, 'Nova', '2025-07-21', 1, 10, 3);
answer: |
  SELECT Player.PlayerName, COUNT(GameCharacter.CharacterID) AS TotalCharacters
  FROM GameCharacter
  INNER JOIN Player ON GameCharacter.PlayerID = Player.PlayerID
  GROUP BY Player.PlayerID, Player.PlayerName
  ORDER BY Player.PlayerID DESC;

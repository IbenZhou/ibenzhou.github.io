prompt: |
  Return the **number of items** bought by the user whose username is **KAT123**.
seed: |
  CREATE TABLE AppUser (
    Username VARCHAR(20) PRIMARY KEY,
    Password VARCHAR(30),
    DateOfBirth DATE
  );
  CREATE TABLE GameCharacter (
    CharacterName VARCHAR(30),
    CharacterID VARCHAR(4) PRIMARY KEY,
    Username VARCHAR(20) REFERENCES AppUser(Username),
    Level INTEGER,
    Money REAL
  );
  CREATE TABLE Character_Item (
    CharacterID VARCHAR(4) REFERENCES GameCharacter(CharacterID),
    ItemName VARCHAR(30)
  );
  INSERT INTO AppUser VALUES
    ('KAT123', 'pass1', '2005-01-09'),
    ('ROB77', 'pass2', '2004-11-30');
  INSERT INTO GameCharacter VALUES
    ('Kat', '0001', 'KAT123', 2, 1500),
    ('Kit', '0002', 'KAT123', 5, 300),
    ('Rex', '0003', 'ROB77', 7, 9000);
  INSERT INTO Character_Item VALUES
    ('0001', 'sword'),
    ('0001', 'shield'),
    ('0002', 'potion'),
    ('0003', 'sword'),
    ('0003', 'map');
answer: |
  SELECT COUNT(Character_Item.ItemName)
  FROM GameCharacter
  INNER JOIN Character_Item ON GameCharacter.CharacterID = Character_Item.CharacterID
  WHERE GameCharacter.Username = 'KAT123';

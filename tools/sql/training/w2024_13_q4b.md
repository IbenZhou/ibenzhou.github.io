schema: |
  SALE(SaleID, BatchID, CustomerID, Quantity, Date)
prompt: |
  Return the **total quantity** of ice cream sold to the customer with the ID **0034E**
  during **2023**.
seed: |
  CREATE TABLE Sale (
    SaleID INTEGER PRIMARY KEY,
    BatchID VARCHAR(8),
    CustomerID VARCHAR(8),
    Quantity INTEGER,
    Date DATE
  );
  INSERT INTO Sale VALUES
    (1, 'B100', '0034E', 40, '2023-02-11'),
    (2, 'B101', '0034E', 25, '2023-08-30'),
    (3, 'B100', '0034E', 30, '2024-01-09'),
    (4, 'B102', '7777X', 60, '2023-05-02');
answer: |
  SELECT SUM(Quantity)
  FROM Sale
  WHERE CustomerID = '0034E' AND Date >= '2023-01-01' AND Date <= '2023-12-31';
wrong: |
  SELECT SUM(Quantity)
  FROM Sale
  WHERE CustomerID = '0034E';

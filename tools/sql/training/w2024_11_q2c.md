title: Amount owed to one supplier

coverage: SELECT, SUM, WHERE
schema: |
  INVOICE(InvoiceID, SupplierID, AmountDue, Paid, DatePaid)
prompt: |
  Return the **total amount due** to supplier **JK675** across all invoices that have
  **not** been paid.
seed: |
  CREATE TABLE Invoice (
    InvoiceID INTEGER PRIMARY KEY,
    SupplierID VARCHAR(8),
    AmountDue REAL,
    Paid BOOLEAN,
    DatePaid DATE
  );
  INSERT INTO Invoice VALUES
    (1, 'JK675', 120.00, FALSE, NULL),
    (2, 'JK675', 80.50, TRUE, '2024-02-03'),
    (3, 'JK675', 44.00, FALSE, NULL),
    (4, 'AB120', 300.00, FALSE, NULL);
answer: |
  SELECT SUM(AmountDue)
  FROM Invoice
  WHERE SupplierID = 'JK675' AND Paid = FALSE;
wrong: |
  SELECT SUM(AmountDue)
  FROM Invoice
  WHERE SupplierID = 'JK675';
check: rows

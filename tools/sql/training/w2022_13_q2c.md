prompt: |
  An astronomy supplier records which company owns each telescope. Return the **total
  number of telescopes** owned by the company whose ID **begins with HW**.
seed: |
  CREATE TABLE Company (
    TelephoneNumber VARCHAR(20),
    CompanyID VARCHAR(8) PRIMARY KEY,
    CompanyName VARCHAR(40)
  );
  CREATE TABLE Telescope (
    TelescopeID INTEGER PRIMARY KEY,
    CompanyID VARCHAR(8) REFERENCES Company(CompanyID),
    SerialNumber VARCHAR(20)
  );
  INSERT INTO Company VALUES
    ('555-0201', 'HW001', 'Harbour Watch'),
    ('555-0202', 'HW777', 'Hill West Optics'),
    ('555-0203', 'NX400', 'Northstar');
  INSERT INTO Telescope VALUES
    (1, 'HW001', 'SN-1001'),
    (2, 'HW001', 'SN-1002'),
    (3, 'HW777', 'SN-2001'),
    (4, 'NX400', 'SN-3001');
answer: |
  SELECT COUNT(TelescopeID)
  FROM Telescope
  WHERE CompanyID LIKE 'HW%';
wrong: |
  SELECT COUNT(TelescopeID)
  FROM Telescope
  WHERE CompanyID = 'HW%';
check: rows

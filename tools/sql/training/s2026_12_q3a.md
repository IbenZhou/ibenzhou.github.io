prompt: |
  The script below was meant to create the table **GameCharacter**. It contains **four**
  errors. Write the corrected CREATE TABLE statement.

  ```
  01 CREATE TABLE GameCharacter(
  02   CharacterID INT,
  03   CharacterName VARCHAR,
  04   DateCreated DATE,
  05   TheLevel INT,
  06   NumberOfPoints INT,
  07   PlayerID VARCHAR,
  08   PRIMARY KEY CharacterID,
  09   FOREIGN KEY(PlayerID)
  10 );
  ```
seed: |
  CREATE TABLE Player (
    PlayerID INTEGER PRIMARY KEY,
    PlayerName VARCHAR(30),
    DateJoined DATE
  );
answer: |
  CREATE TABLE GameCharacter (
    CharacterID INTEGER PRIMARY KEY,
    CharacterName VARCHAR(30),
    DateCreated DATE,
    TheLevel INTEGER,
    NumberOfPoints INTEGER,
    PlayerID INTEGER,
    FOREIGN KEY (PlayerID) REFERENCES Player(PlayerID)
  );
wrong: |
  CREATE TABLE GameCharacter (
    CharacterID INT,
    CharacterName VARCHAR,
    DateCreated DATE,
    TheLevel INT,
    NumberOfPoints INT,
    PlayerID VARCHAR,
    PRIMARY KEY CharacterID,
    FOREIGN KEY(PlayerID)
  );

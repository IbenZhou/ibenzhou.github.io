prompt: |
  A toy library lends copies of toys to customers. Return each customer's **first name**
  and **ID**, plus the **total number of days** they have borrowed toys for. Give the
  total a suitable name.
seed: |
  CREATE TABLE Customer (
    CustomerID VARCHAR(6) PRIMARY KEY,
    ContactFirstName VARCHAR(30),
    Phone VARCHAR(20)
  );
  CREATE TABLE Borrow (
    BorrowID VARCHAR(6) PRIMARY KEY,
    CustomerID VARCHAR(6) REFERENCES Customer(CustomerID),
    UniqueToyID VARCHAR(8),
    StartDate DATE,
    NumberDays INTEGER
  );
  INSERT INTO Customer VALUES
    ('C001', 'Leila', '555-0110'),
    ('C002', 'Jonas', '555-0111'),
    ('C003', 'Mei', '555-0112');
  INSERT INTO Borrow VALUES
    ('A2226', 'C001', 'T900', '2026-01-05', 7),
    ('A2227', 'C001', 'T901', '2026-01-20', 3),
    ('A2228', 'C002', 'T900', '2026-01-21', 14),
    ('A2229', 'C003', 'T902', '2026-02-02', 5);
answer: |
  SELECT CUSTOMER.ContactFirstName, CUSTOMER.CustomerID, SUM(Borrow.NumberDays) AS TotalDays
  FROM Borrow
  INNER JOIN Customer ON Customer.CustomerID = Borrow.CustomerID
  GROUP BY CUSTOMER.CustomerID, CUSTOMER.ContactFirstName;

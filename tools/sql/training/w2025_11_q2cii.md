title: Count one student's placements at one company
marks: 4
difficulty: advanced
source: 同构于 2025 冬季 P11 Q2(c)(ii)
coverage: SELECT, COUNT, WHERE
schema: |
  PLACEMENT(PlacementID, StudentID, CompanyID, StartDate, EndDate, Complete)
prompt: |
  Return the **number of placements** that student **LDEA01** has completed at company
  **NEAM**. Give the count a suitable name.
seed: |
  CREATE TABLE Placement (
    PlacementID INTEGER PRIMARY KEY,
    StudentID VARCHAR(8),
    CompanyID VARCHAR(8),
    StartDate DATE,
    EndDate DATE,
    Complete BOOLEAN
  );
  INSERT INTO Placement VALUES
    (1, 'LDEA01', 'NEAM', '2024-06-01', '2024-06-28', TRUE),
    (2, 'LDEA01', 'NEAM', '2025-01-06', '2025-02-14', FALSE),
    (3, 'LDEA01', 'HALO', '2024-09-02', '2024-09-27', TRUE),
    (4, 'BRX777', 'NEAM', '2024-09-09', '2024-10-04', TRUE);
answer: |
  SELECT COUNT(PlacementID) AS CompletedPlacements
  FROM Placement
  WHERE StudentID = 'LDEA01' AND CompanyID = 'NEAM' AND Complete = TRUE;
wrong: |
  SELECT COUNT(PlacementID) AS CompletedPlacements
  FROM Placement
  WHERE StudentID = 'LDEA01' AND CompanyID = 'NEAM';
check: rows

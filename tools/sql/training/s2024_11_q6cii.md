    EventID INTEGER PRIMARY KEY,
    EventName VARCHAR(40),
    EventDate DATE,
    PlayerID INTEGER
  );
  INSERT INTO Event VALUES
    (1, 'Sprint', '2024-03-02', 11),
    (2, 'Long jump', '2024-03-02', 11),

    (4, 'Sprint', '2024-03-16', 13),
    (5, 'Relay', '2024-03-16', 11);

  SELECT PlayerID, COUNT(EventID)
  FROM Event
  GROUP BY PlayerID;

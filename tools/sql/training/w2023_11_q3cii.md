prompt: |
  Return the **total quantity of items** ordered by the customer whose ID is **HJ231**.
seed: |
  CREATE TABLE Customer (
    CustomerID VARCHAR(8) PRIMARY KEY,
    FirstName VARCHAR(30),
    LastName VARCHAR(30),
    Town VARCHAR(30)
  );
  CREATE TABLE Shop_Order (
    OrderNo INTEGER PRIMARY KEY,
    CustomerID VARCHAR(8) REFERENCES Customer(CustomerID),
    OrderDate DATE
  );
  CREATE TABLE Order_Item (
    ItemNumber VARCHAR(8),
    OrderNo INTEGER REFERENCES Shop_Order(OrderNo),
    Quantity INTEGER,
    PRIMARY KEY (ItemNumber, OrderNo)
  );
  INSERT INTO Customer VALUES
    ('HJ231', 'Hilda', 'Jensen', 'Bath'),
    ('PP900', 'Piotr', 'Pawlak', 'Leeds');
  INSERT INTO Shop_Order VALUES
    (1, 'HJ231', '2023-10-02'),
    (2, 'HJ231', '2023-10-09'),
    (3, 'PP900', '2023-10-11');
  INSERT INTO Order_Item VALUES
    ('I100', 1, 4),
    ('I205', 1, 2),
    ('I100', 2, 6),
    ('I100', 3, 9);
answer: |
  SELECT SUM(Order_Item.Quantity)
  FROM Order_Item
  INNER JOIN Shop_Order ON Order_Item.OrderNo = Shop_Order.OrderNo
  WHERE Shop_Order.CustomerID = 'HJ231';
wrong: |
  SELECT SUM(Order_Item.Quantity)
  FROM Order_Item
  INNER JOIN Shop_Order ON Order_Item.OrderNo = Shop_Order.OrderNo;

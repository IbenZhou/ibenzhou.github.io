prompt: |
  Return each customer's name together with the **number of orders** they have not yet
  collected. Include only customers who have at least one uncollected order, and give
  the count a suitable name.
seed: |
  CREATE TABLE Customer (
    CustomerID INTEGER PRIMARY KEY,
    CustomerName VARCHAR(40),
    Email VARCHAR(60)
  );
  CREATE TABLE Orders (
    OrderID INTEGER PRIMARY KEY,
    CustomerID INTEGER REFERENCES Customer(CustomerID),
    Date DATE,
    Collected BOOLEAN
  );
  INSERT INTO Customer VALUES
    (1, 'Nadia Farah', 'nadia@example.org'),
    (2, 'Peter Lang', 'peter@example.org'),
    (3, 'Ruth Okoro', 'ruth@example.org');
  INSERT INTO Orders VALUES
    (100, 1, '2025-03-01', FALSE),
    (101, 1, '2025-03-04', TRUE),
    (102, 1, '2025-03-08', FALSE),
    (103, 2, '2025-03-09', TRUE),
    (104, 3, '2025-03-11', FALSE);
answer: |
  SELECT Customer.CustomerName, COUNT(Orders.OrderID) AS UncollectedOrders
  FROM Customer
  INNER JOIN Orders ON Customer.CustomerID = Orders.CustomerID
  WHERE Orders.Collected = FALSE
  GROUP BY Customer.CustomerName;

prompt: |
  Return the **number of containers** currently loaded on the ship named **Caledonia**.
seed: |
  CREATE TABLE Ship (
    ShipID VARCHAR(8) PRIMARY KEY,
    Type VARCHAR(20),
    Capacity INTEGER,
    ShipName VARCHAR(30)
  );
  CREATE TABLE Container (
    ContainerID VARCHAR(10) PRIMARY KEY,
    Type VARCHAR(20),
    Weight REAL,
    OwnerName VARCHAR(40),
    ShipID VARCHAR(8) REFERENCES Ship(ShipID)
  );
  INSERT INTO Ship VALUES
    ('S1', 'cargo', 900, 'Caledonia'),
    ('S2', 'cargo', 700, 'Aurora');
  INSERT INTO Container VALUES
    ('C100', 'dry', 2200.5, 'Harbour Freight', 'S1'),
    ('C101', 'reefer', 3100.0, 'Northline', 'S1'),
    ('C102', 'dry', 1800.0, 'Harbour Freight', 'S2');
answer: |
  SELECT COUNT(Container.ContainerID)
  FROM Container
  INNER JOIN Ship ON Container.ShipID = Ship.ShipID
  WHERE Ship.ShipName = 'Caledonia';
wrong: |
  SELECT COUNT(Container.ContainerID)
  FROM Container
  INNER JOIN Ship ON Container.ShipID = Ship.ShipID;
